
inventory = {
    "laptop": {
        "price": 999.99, "quantity": 15
    },
    "mouse": {
        "price": 29.99, "quantity": 50
    },
    "keyboard": {
        "price": 49.99, "quantity": 30
    },
    "mousepad": {
        "price": 9.99, "quantity": 20
        }, 
}

#Display inventory
def display_inventory():
    """Print all inventory in a formatted way."""
    print("\n" + "=" * 30)
    print("          Inventory")
    print("=" * 30) 

    if not inventory:
        print("No inventory yet.")
        return
    for key, info in inventory.items():
        print(f" {key.capitalize()}: {info['price']}, {info['quantity']}")
        print("=" * 30)

    print(f"\nTotal inventory: {len(inventory)}")
    print("=" * 30)

display_inventory()

#look up inventory

search = input("\nLook up an item in inventory: ").lower()

inventoryy = inventory.get(search)

if inventoryy:
    print(f"\nFound: {search.capitalize()}")
    print(f"\nPrice: {inventoryy['price']}, Quantity: {inventoryy['quantity']}")
else: 
    print(f"No inventory found for '{search}'.")
